class CuentaActivity : AppCompatActivity() {

    private lateinit var binding: ActivityCuentaBinding
    private lateinit var pedidoDao: PedidoDao
    private var idMesa: Int = -1
    private var numeroMesa: Int = -1
    private var pedidoActual: Pedido? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        binding = ActivityCuentaBinding.inflate(layoutInflater)
        setContentView(binding.root)

        pedidoDao = PedidoDao(this)

        idMesa = intent.getIntExtra("EXTRA_MESA_ID", -1)
        numeroMesa = intent.getIntExtra("EXTRA_MESA_NUMERO", -1)

        binding.tvTituloCuenta.text = "Cuenta Â· Mesa $numeroMesa"
        binding.recyclerDetalleCuenta.layoutManager = LinearLayoutManager(this)

        cargarCuenta()

        // BotÃ³n Cerrar Cuenta en una transacciÃ³n (HU-09, CA2)
        binding.btnCerrarCuenta.setOnClickListener {
            val pedido = pedidoActual
            if (pedido == null) {
                Toast.makeText(this, "No hay pedido abierto", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            val exito = pedidoDao.cerrarCuenta(pedido.id, idMesa)
            if (exito) {
                Toast.makeText(this, "Mesa $numeroMesa liberada", Toast.LENGTH_LONG).show()
                binding.btnCerrarCuenta.isEnabled = false
                finish() // Cierra la pantalla y regresa
            } else {
                Toast.makeText(this, "Error al cerrar la cuenta", Toast.LENGTH_SHORT).show()
            }
        }

        // Compartir la cuenta por WhatsApp / Apps de mensajerÃ­a (HU-11, CA1, CA2, CA3)
        binding.btnCompartirWhatsApp.setOnClickListener {
            val pedido = pedidoActual
            val sb = StringBuilder()
            sb.append("PollerÃ­a El Buen Sabor\n")
            sb.append("Cuenta - Mesa $numeroMesa\n")

            if (pedido != null) {
                sb.append("Pedido #${pedido.id} Â· ${pedido.fecha}\n\n")
                val detalles = pedidoDao.listarDetallesPedido(pedido.id)
                for (d in detalles) {
                    sb.append("${d.cantidad} x ${d.nombrePlato} - S/ %.2f\n".format(d.subtotal))
                }
                sb.append("\nTotal: S/ %.2f\n".format(pedido.total))
            } else {
                sb.append("Total: S/ 0.00\n")
            }
            sb.append("Â¡Gracias por su preferencia!")

            val shareIntent = Intent(Intent.ACTION_SEND).apply {
                type = "text/plain"
                putExtra(Intent.EXTRA_TEXT, sb.toString())
            }
            startActivity(Intent.createChooser(shareIntent, "Compartir cuenta"))
        }
    }

    private fun cargarCuenta() {
        val pedido = pedidoDao.obtenerPedidoAbiertoPorMesa(idMesa)
        pedidoActual = pedido

        if (pedido != null) {
            binding.tvSubtituloCuenta.text = "Pedido #${pedido.id} Â· ${pedido.fecha}"
            val detalles = pedidoDao.listarDetallesPedido(pedido.id)
            binding.recyclerDetalleCuenta.adapter = DetallePedidoAdapter(detalles)
            binding.tvTotalCuenta.text = "S/ %.2f".format(pedido.total)
            binding.btnCerrarCuenta.isEnabled = true
        } else {
            binding.tvSubtituloCuenta.text = "No hay pedido abierto para esta mesa"
            binding.recyclerDetalleCuenta.adapter = DetallePedidoAdapter(emptyList())
            binding.tvTotalCuenta.text = "S/ 0.00"
            // CA3: Si la mesa estÃ¡ LIBRE o sin pedido, el botÃ³n estÃ¡ deshabilitado
            binding.btnCerrarCuenta.isEnabled = false
        }
    }
}

import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper

class DBHelper(context: Context) : SQLiteOpenHelper(context, DB_NAME, null, DB_VERSION) {

    companion object {
        const val DB_NAME = "saborapp.db"
        const val DB_VERSION = 2
    }

    override fun onCreate(db: SQLiteDatabase) {
        // Tablas de CatÃ¡logo y Usuarios
        db.execSQL("CREATE TABLE usuario (id INTEGER PRIMARY KEY AUTOINCREMENT, usuario TEXT UNIQUE, clave TEXT, rol TEXT)")
        db.execSQL("CREATE TABLE plato (id INTEGER PRIMARY KEY AUTOINCREMENT, nombre TEXT NOT NULL, categoria TEXT, precio REAL CHECK(precio > 0), disponible INTEGER)")
        db.execSQL("CREATE TABLE mesa (id INTEGER PRIMARY KEY AUTOINCREMENT, numero INTEGER UNIQUE, capacidad INTEGER, estado TEXT)")

        // Tablas de OperaciÃ³n Principal (Sprint 3)
        db.execSQL("CREATE TABLE pedido (id INTEGER PRIMARY KEY AUTOINCREMENT, id_mesa INTEGER REFERENCES mesa(id), fecha TEXT, estado TEXT, total REAL)")
        db.execSQL("CREATE TABLE detalle_pedido (id INTEGER PRIMARY KEY AUTOINCREMENT, id_pedido INTEGER REFERENCES pedido(id) ON DELETE CASCADE, id_plato INTEGER REFERENCES plato(id), cantidad INTEGER CHECK(cantidad > 0), precio_unit REAL, subtotal REAL)")

        // Usuarios por defecto (HU-04)
        db.execSQL("INSERT INTO usuario (usuario, clave, rol) VALUES ('admin', '1234', 'ADMIN')")
        db.execSQL("INSERT INTO usuario (usuario, clave, rol) VALUES ('mozo', '1234', 'MOZO')")
    }

    override fun onUpgrade(db: SQLiteDatabase, oldVersion: Int, newVersion: Int) {
        if (oldVersion < 2) {
            db.execSQL("CREATE TABLE pedido (id INTEGER PRIMARY KEY AUTOINCREMENT, id_mesa INTEGER REFERENCES mesa(id), fecha TEXT, estado TEXT, total REAL)")
            db.execSQL("CREATE TABLE detalle_pedido (id INTEGER PRIMARY KEY AUTOINCREMENT, id_pedido INTEGER REFERENCES pedido(id) ON DELETE CASCADE, id_plato INTEGER REFERENCES plato(id), cantidad INTEGER CHECK(cantidad > 0), precio_unit REAL, subtotal REAL)")
        }
    }

    override fun onConfigure(db: SQLiteDatabase) {
        super.onConfigure(db)
        db.setForeignKeyConstraintsEnabled(true)
    }
}

import android.content.ContentValues
import android.content.Context
import android.database.Cursor
import android.database.sqlite.SQLiteConstraintException

class MesaDao(context: Context) {
    private val dbHelper = DBHelper(context)

    // Registrar una mesa validando si el nÃºmero ya existe (HU-06, CA1)
    fun insertar(mesa: Mesa): String {
        val db = dbHelper.writableDatabase
        return try {
            val values = ContentValues().apply {
                put("numero", mesa.numero)
                put("capacidad", mesa.capacidad)
                put("estado", "LIBRE") // Toda mesa nueva nace en estado LIBRE (CA3)
            }
            val resultado = db.insert("mesa", null, values)
            db.close()
            if (resultado != -1L) "OK" else "Error al registrar la mesa"
        } catch (e: SQLiteConstraintException) {
            db.close()
            "La mesa ya existe" // Criterio de aceptaciÃ³n HU-06
        }
    }

    // Listar las mesas ordenadas por su nÃºmero
    fun listar(): List<Mesa> {
        val lista = mutableListOf<Mesa>()
        val db = dbHelper.readableDatabase
        val cursor: Cursor = db.rawQuery("select * from mesa order by numero", null)

        if (cursor.moveToFirst()) {
            do {
                val id = cursor.getInt(cursor.getColumnIndexOrThrow("id"))
                val numero = cursor.getInt(cursor.getColumnIndexOrThrow("numero"))
                val capacidad = cursor.getInt(cursor.getColumnIndexOrThrow("capacidad"))
                val estado = cursor.getString(cursor.getColumnIndexOrThrow("estado"))

                lista.add(Mesa(id, numero, capacidad, estado))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }
}

    private var mesaSeleccionada: Mesa? = null
    private var platosDisponibles: List<Plato> = emptyList()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        binding = ActivityPedidoBinding.inflate(layoutInflater)
        setContentView(binding.root)

        mesaDao = MesaDao(this)
        platoDao = PlatoDao(this)
        pedidoDao = PedidoDao(this)

        binding.recyclerMesasPedido.layoutManager = GridLayoutManager(this, 3)
        binding.recyclerDetallePedido.layoutManager = LinearLayoutManager(this)

        cargarSpinnerPlatos()

        // BotÃ³n Agregar Plato
        binding.btnAgregarPlato.setOnClickListener {
            val mesa = mesaSeleccionada
            if (mesa == null) {
                Toast.makeText(this, "Seleccione una mesa", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            val posPlato = binding.spPlatosDisponibles.selectedItemPosition
            if (posPlato < 0 || posPlato >= platosDisponibles.size) {
                Toast.makeText(this, "Seleccione un plato disponible", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }
            val plato = platosDisponibles[posPlato]

            val cantStr = binding.etCantidadPlato.text.toString().trim()
            val cantidad = cantStr.toIntOrNull()
            if (cantidad == null || cantidad <= 0) {
                binding.etCantidadPlato.error = "Cantidad invÃ¡lida"
                return@setOnClickListener
            }

            // TransacciÃ³n: Guardar plato en el pedido (HU-08, CA2, CA3, CA4)
            val exito = pedidoDao.agregarPlatoAPedido(mesa.id, plato.id, cantidad)
            if (exito) {
                Toast.makeText(this, "Plato agregado", Toast.LENGTH_SHORT).show()
                binding.etCantidadPlato.setText("1")
                cargarMesas()
                seleccionarMesa(mesaDao.listar().find { it.id == mesa.id } ?: mesa)
            } else {
                Toast.makeText(this, "Error al agregar el plato", Toast.LENGTH_SHORT).show()
            }
        }

        // BotÃ³n Ver Cuenta (HU-09, CA1)

class PedidoDao(context: Context) {
    private val dbHelper = DBHelper(context)

    // Obtener pedido ABIERTO para una mesa determinada
    fun obtenerPedidoAbiertoPorMesa(idMesa: Int): Pedido? {
        val db = dbHelper.readableDatabase
        val cursor: Cursor = db.rawQuery(
            "SELECT * FROM pedido WHERE id_mesa = ? AND estado = 'ABIERTO'",
            arrayOf(idMesa.toString())
        )
        var pedido: Pedido? = null
        if (cursor.moveToFirst()) {
            val id = cursor.getInt(cursor.getColumnIndexOrThrow("id"))
            val fecha = cursor.getString(cursor.getColumnIndexOrThrow("fecha"))
            val estado = cursor.getString(cursor.getColumnIndexOrThrow("estado"))
            val total = cursor.getDouble(cursor.getColumnIndexOrThrow("total"))
            pedido = Pedido(id, idMesa, fecha, estado, total)
        }
        cursor.close()
        db.close()
        return pedido
    }

    // Agregar plato al pedido usando una transacciÃ³n de SQLite (HU-08, CA2, CA3, CA4)
    fun agregarPlatoAPedido(idMesa: Int, idPlato: Int, cantidad: Int): Boolean {
        val db = dbHelper.writableDatabase
        db.beginTransaction()
        try {
            // 1. Obtener precio actual del plato
            val cursorPlato = db.rawQuery("SELECT precio FROM plato WHERE id = ?", arrayOf(idPlato.toString()))
            if (!cursorPlato.moveToFirst()) {
                cursorPlato.close()
                return false
            }
            val precioUnit = cursorPlato.getDouble(0)
            cursorPlato.close()

            // 2. Obtener o crear el pedido ABIERTO para la mesa
            var idPedido: Long = -1
            val cursorPedido = db.rawQuery(
                "SELECT id FROM pedido WHERE id_mesa = ? AND estado = 'ABIERTO'",
                arrayOf(idMesa.toString())
            )
            if (cursorPedido.moveToFirst()) {
                idPedido = cursorPedido.getLong(0)
            } else {
                val fechaActual = SimpleDateFormat("dd/MM/yyyy HH:mm", Locale.getDefault()).format(Date())
                val valPedido = ContentValues().apply {
                    put("id_mesa", idMesa)
                    put("fecha", fechaActual)
                    put("estado", "ABIERTO")
                    put("total", 0.0)
                }
                idPedido = db.insert("pedido", null, valPedido)

                // Cambiar estado de la mesa a OCUPADA
                val valMesa = ContentValues().apply { put("estado", "OCUPADA") }
                db.update("mesa", valMesa, "id = ?", arrayOf(idMesa.toString()))
            }
            cursorPedido.close()

            if (idPedido == -1L) return false

            // 3. Insertar o actualizar item en detalle_pedido
            val subtotal = precioUnit * cantidad
            val cursorDetalle = db.rawQuery(
                "SELECT id, cantidad FROM detalle_pedido WHERE id_pedido = ? AND id_plato = ?",
                arrayOf(idPedido.toString(), idPlato.toString())
            )
            if (cursorDetalle.moveToFirst()) {
                val idDetalle = cursorDetalle.getLong(0)
                val cantExistente = cursorDetalle.getInt(1)
                val nuevaCant = cantExistente + cantidad
                val nuevoSubtotal = precioUnit * nuevaCant
                val valDetalle = ContentValues().apply {
                    put("cantidad", nuevaCant)
                    put("subtotal", nuevoSubtotal)
                }
                db.update("detalle_pedido", valDetalle, "id = ?", arrayOf(idDetalle.toString()))
            } else {
                val valDetalle = ContentValues().apply {
                    put("id_pedido", idPedido)
                    put("id_plato", idPlato)
                    put("cantidad", cantidad)
                    put("precio_unit", precioUnit)
                    put("subtotal", subtotal)
                }
                db.insert("detalle_pedido", null, valDetalle)
            }
            cursorDetalle.close()

            // 4. Recalcular total del pedido
            val cursorTotal = db.rawQuery(
                "SELECT SUM(subtotal) FROM detalle_pedido WHERE id_pedido = ?",
                arrayOf(idPedido.toString())
            )
            var nuevoTotal = 0.0
            if (cursorTotal.moveToFirst()) {
                nuevoTotal = cursorTotal.getDouble(0)
            }
            cursorTotal.close()

            val valTotal = ContentValues().apply { put("total", nuevoTotal) }
            db.update("pedido", valTotal, "id = ?", arrayOf(idPedido.toString()))

            db.setTransactionSuccessful()
            return true
        } catch (e: Exception) {
            return false
        } finally {
            db.endTransaction()
            db.close()
        }
    }

    // Listar detalles del pedido con JOIN a la tabla plato
    fun listarDetallesPedido(idPedido: Int): List<DetallePedido> {
        val lista = mutableListOf<DetallePedido>()
        val db = dbHelper.readableDatabase
        val query = """
            SELECT d.id, d.id_pedido, d.id_plato, p.nombre, d.cantidad, d.precio_unit, d.subtotal 
            FROM detalle_pedido d 
            INNER JOIN plato p ON d.id_plato = p.id 
            WHERE d.id_pedido = ?
        """.trimIndent()

        val cursor: Cursor = db.rawQuery(query, arrayOf(idPedido.toString()))

        if (cursor.moveToFirst()) {
            do {
                val id = cursor.getInt(cursor.getColumnIndexOrThrow("id"))
                val idPlato = cursor.getInt(cursor.getColumnIndexOrThrow("id_plato"))
                val nombrePlato = cursor.getString(cursor.getColumnIndexOrThrow("nombre"))
                val cantidad = cursor.getInt(cursor.getColumnIndexOrThrow("cantidad"))
                val precioUnit = cursor.getDouble(cursor.getColumnIndexOrThrow("precio_unit"))
                val subtotal = cursor.getDouble(cursor.getColumnIndexOrThrow("subtotal"))

                lista.add(DetallePedido(id, idPedido, idPlato, nombrePlato, cantidad, precioUnit, subtotal))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

    // Cerrar la cuenta de la mesa en una sola transacciÃ³n (HU-09, CA2)
    fun cerrarCuenta(idPedido: Int, idMesa: Int): Boolean {
        val db = dbHelper.writableDatabase
        db.beginTransaction()
        try {
            val valPedido = ContentValues().apply { put("estado", "CERRADO") }
            db.update("pedido", valPedido, "id = ?", arrayOf(idPedido.toString()))

            val valMesa = ContentValues().apply { put("estado", "LIBRE") }
            db.update("mesa", valMesa, "id = ?", arrayOf(idMesa.toString()))

            db.setTransactionSuccessful()
            return true
        } catch (e: Exception) {
            return false
        } finally {
            db.endTransaction()
            db.close()
        }
    }
}

class PlatoDao(context: Context) {
    private val dbHelper = DBHelper(context)

    fun insertar(plato: Plato): Boolean {
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put("nombre", plato.nombre)
            put("categoria", plato.categoria)
            put("precio", plato.precio)
            put("disponible", plato.disponible)
        }
        val resultado = db.insert("plato", null, values)
        db.close()
        return resultado != -1L
    }

    fun actualizar(plato: Plato): Boolean {
        val db = dbHelper.writableDatabase
        val values = ContentValues().apply {
            put("nombre", plato.nombre)
            put("categoria", plato.categoria)
            put("precio", plato.precio)
            put("disponible", plato.disponible)
        }
        val filas = db.update("plato", values, "id = ?", arrayOf(plato.id.toString()))
        db.close()
        return filas > 0
    }

    fun eliminar(id: Int): String {
        val db = dbHelper.writableDatabase
        // Verificar antes si el plato estÃ¡ referenciado en un detalle de pedido
        val cursor = db.rawQuery("SELECT COUNT(*) FROM detalle_pedido WHERE id_plato = ?", arrayOf(id.toString()))
        if (cursor.moveToFirst() && cursor.getInt(0) > 0) {
            cursor.close()
            db.close()
            return "No se puede eliminar: tiene pedidos"
        }
        cursor.close()

        return try {
            val filas = db.delete("plato", "id = ?", arrayOf(id.toString()))
            db.close()
            if (filas > 0) "OK" else "Error al eliminar"
        } catch (e: SQLiteConstraintException) {
            db.close()
            "No se puede eliminar: tiene pedidos"
        } catch (e: Exception) {
            db.close()
            "No se puede eliminar: tiene pedidos"
        }
    }

    fun obtener(id: Int): Plato? {
        val db = dbHelper.readableDatabase
        val cursor: Cursor = db.rawQuery("SELECT * FROM plato WHERE id = ?", arrayOf(id.toString()))
        var plato: Plato? = null
        if (cursor.moveToFirst()) {
            val nombre = cursor.getString(cursor.getColumnIndexOrThrow("nombre"))
            val categoria = cursor.getString(cursor.getColumnIndexOrThrow("categoria"))
            val precio = cursor.getDouble(cursor.getColumnIndexOrThrow("precio"))
            val disponible = cursor.getInt(cursor.getColumnIndexOrThrow("disponible"))
            plato = Plato(id, nombre, categoria, precio, disponible)
        }
        cursor.close()
        db.close()
        return plato
    }

    // Listar platos con filtro por nombre y categorÃ­a (HU-07, CA3)
    fun listar(filtroNombre: String = "", categoriaFiltro: String = "Todos"): List<Plato> {
        val lista = mutableListOf<Plato>()
        val db = dbHelper.readableDatabase

        var query = "SELECT * FROM plato WHERE 1=1"
        val params = mutableListOf<String>()

        if (filtroNombre.isNotBlank()) {
            query += " AND nombre LIKE ?"
            params.add("%$filtroNombre%")
        }

        if (categoriaFiltro != "Todos" && categoriaFiltro.isNotBlank()) {
            query += " AND categoria = ?"
            params.add(categoriaFiltro)
        }

        query += " ORDER BY categoria, nombre"

        val cursor: Cursor = db.rawQuery(query, params.toTypedArray())

        if (cursor.moveToFirst()) {
            do {
                val id = cursor.getInt(cursor.getColumnIndexOrThrow("id"))
                val nombre = cursor.getString(cursor.getColumnIndexOrThrow("nombre"))
                val categoria = cursor.getString(cursor.getColumnIndexOrThrow("categoria"))
                val precio = cursor.getDouble(cursor.getColumnIndexOrThrow("precio"))
                val disponible = cursor.getInt(cursor.getColumnIndexOrThrow("disponible"))

                lista.add(Plato(id, nombre, categoria, precio, disponible))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

    // Listar Ãºnicamente los platos disponibles (HU-07, CA4)
    fun listarDisponibles(): List<Plato> {
        val lista = mutableListOf<Plato>()

                val categoria = cursor.getString(cursor.getColumnIndexOrThrow("categoria"))
                val precio = cursor.getDouble(cursor.getColumnIndexOrThrow("precio"))

                lista.add(Plato(id, nombre, categoria, precio, 1))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }

            SELECT p.nombre, SUM(d.cantidad) AS total_cant 
            FROM detalle_pedido d 
            INNER JOIN pedido pe ON d.id_pedido = pe.id 
            INNER JOIN plato p ON d.id_plato = p.id 
            WHERE pe.estado = 'CERRADO' AND substr(pe.fecha, 1, 10) = ? 
            GROUP BY p.id, p.nombre 
            ORDER BY total_cant DESC 
            LIMIT 5
        """.trimIndent()

        val cursor: Cursor = db.rawQuery(query, arrayOf(fechaHoy))
        if (cursor.moveToFirst()) {
            do {
                val nombre = cursor.getString(cursor.getColumnIndexOrThrow("nombre"))
                val cantidad = cursor.getInt(cursor.getColumnIndexOrThrow("total_cant"))
                lista.add(TopPlato(nombre, cantidad))
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return lista
    }
}

class ReportesActivity : AppCompatActivity() {

        binding.recyclerTopPlatos.layoutManager = LinearLayoutManager(this)

            val ticketPromedio = reporteDao.ticketPromedioDelDia()
            val topPlatos = reporteDao.topPlatosDelDia()

            sb.append("Total Pedidos: $totalPedidos\n")
            sb.append("Ticket Promedio: S/ %.2f\n\n".format(ticketPromedio))
            sb.append("Platos mÃ¡s pedidos:\n")

                sb.append("Sin ventas hoy")
            } else {
                for (p in topPlatos) {
                    sb.append("- ${p.nombre}: ${p.cantidad} und.\n")
                }

                type = "text/plain"
                putExtra(Intent.EXTRA_TEXT, sb.toString())

            startActivity(Intent.createChooser(shareIntent, "Compartir resumen de ventas"))

    private fun cargarReporte() {
        val ventas = reporteDao.ventaDelDia()
        val pedidos = reporteDao.totalPedidosDelDia()
        val ticketPromedio = reporteDao.ticketPromedioDelDia()
        val topPlatos = reporteDao.topPlatosDelDia()

        binding.tvCantidadPedidosHoy.text = pedidos.toString()
        binding.tvTicketPromedio.text = "S/ %.2f".format(ticketPromedio)

        // HU-10, CA3: Si no hay ventas hoy, muestra "Sin ventas hoy"
        if (topPlatos.isEmpty()) {
            binding.tvSinVentasHoy.visibility = View.VISIBLE
            binding.recyclerTopPlatos.adapter = TopPlatoAdapter(emptyList(), 0)
        } else {
            binding.tvSinVentasHoy.visibility = View.GONE
            val maxCantidad = topPlatos.maxOf { it.cantidad }
            binding.recyclerTopPlatos.adapter = TopPlatoAdapter(topPlatos, maxCantidad)
        }
    }
}

import android.content.Context
import android.database.Cursor

class UsuarioDao(context: Context) {
    private val dbHelper = DBHelper(context)

    fun validarUsuario(usuario: String, clave: String): Usuario? {
        val db = dbHelper.readableDatabase
        val cursor: Cursor = db.rawQuery(
            "SELECT * FROM usuario WHERE usuario = ? AND clave = ?",
            arrayOf(usuario, clave)
        )
        var user: Usuario? = null
        if (cursor.moveToFirst()) {
            val id = cursor.getInt(cursor.getColumnIndexOrThrow("id"))
            val userStr = cursor.getString(cursor.getColumnIndexOrThrow("usuario"))
            val passStr = cursor.getString(cursor.getColumnIndexOrThrow("clave"))
            val rolStr = cursor.getString(cursor.getColumnIndexOrThrow("rol"))
            user = Usuario(id, userStr, passStr, rolStr)
        }
        cursor.close()
        db.close()
        return user
    }
